Python Flask Server Bridge (server.py)

Run this Python script on your PC connected to the ESP32 via USB serial COM port:

# Install dependencies: pip install flask flask-cors pyserial
from flask import Flask, jsonify, request
from flask_cors import CORS
import serial
import time

app = Flask(__name__)
CORS(app) # Enable CORS for Dashboard

PORT = "COM5" # Change to your ESP32 serial port
BAUD = 115200

patients = {
    1: {
        "name": "Aarav Patel",
        "age": 24,
        "blood_group": "B+",
        "allergies": "Penicillin",
        "conditions": "Asthma",
        "surgeries": "Appendectomy - 2024",
        "medications": "Salbutamol",
        "emergency_contact": "+91 XXXXX XXXXX"
    },
    2: {
        "name": "Rohan Shah",
        "age": 31,
        "blood_group": "O+",
        "allergies": "None",
        "conditions": "Diabetes",
        "surgeries": "None",
        "medications": "Metformin",
        "emergency_contact": "+91 XXXXX XXXXX"
    }
}

try:
    esp32 = serial.Serial(PORT, BAUD, timeout=1)
    esp32.dtr = False
    esp32.rts = False
    time.sleep(2)
    esp32.reset_input_buffer()
    print("ESP32 Connected on", PORT)
except Exception as e:
    print("Serial Connection Error:", e)
    esp32 = None

@app.route('/api/status', methods=['GET'])
def status():
    return jsonify({"status": "online", "serial_connected": esp32 is not None})

@app.route('/api/patients', methods=['GET'])
def get_patients():
    return jsonify(patients)

@app.route('/api/scan', methods=['POST'])
def scan():
    if not esp32:
        return jsonify({"error": "ESP32 not connected"}), 500
    
    esp32.write(b"SCAN\n")
    start_time = time.time()
    
    while time.time() - start_time < 10:
        line = esp32.readline().decode('utf-8', errors='ignore').strip()
        if line.startswith("FINGERPRINT_ID:"):
            fid = int(line.split(":")[1])
            patient = patients.get(fid, None)
            return jsonify({"status": "found", "id": fid, "patient": patient})
        elif line == "NO_MATCH":
            return jsonify({"status": "no_match"}), 404
            
    return jsonify({"status": "timeout"}), 408

@app.route('/api/enroll', methods=['POST'])
def enroll():
    data = request.json
    fid = data.get('id')
    if not fid or not esp32:
        return jsonify({"error": "Invalid request"}), 400
        
    esp32.write(f"ENROLL:{fid}\n".encode())
    start_time = time.time()
    
    while time.time() - start_time < 15:
        line = esp32.readline().decode('utf-8', errors='ignore').strip()
        if line == "ENROLL_SUCCESS":
            patients[fid] = data.get('patient', {})
            return jsonify({"status": "success", "id": fid})
        elif line == "ENROLL_FAILED":
            return jsonify({"status": "failed"}), 400
            
    return jsonify({"status": "timeout"}), 408

if __name__ == '__main__':
    app.run(host='0.0.0.0', port=5000, debug=True)
